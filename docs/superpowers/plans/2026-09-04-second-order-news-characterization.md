**Goal:** Close the residual scope of #408 — an independent, deterministic detector that labels "second-order/spillover" news headlines (correctly names our ticker's company but the causal driver is a third party), plus a script that counts how often it occurs in `news_log` and reports the realised forward return, joined against the existing `sentiment_signals` columns.

**Architecture:** A pure-function module `src/analysis/second_order_news.py` (classification + aggregation, no I/O — matches the established pattern in `src/analysis/watchlist_coverage.py`), plus a thin orchestrator script `scripts/characterize_second_order_news.py` (does the Postgres I/O, calls the pure functions, writes `docs/evidence/second_order_news.json`). No migration, no schema change, no gate/prompt/config touched — read-only against `news_log`, `ticker_lookup`, `sentiment_signals`, `llm_responses`.

**Tech Stack:** Python 3, psycopg2 (already a project dependency), pytest. No new dependencies.

```python
def nomi_candidati(company_name: str, aliases: list[str]) -> tuple[str, ...]:
    """Tutte le forme sotto cui l'azienda puo' comparire in un titolo.

    Nome completo, alias, e la forma corta di ciascuno (tolto il suffisso
    societario) — deduplicati, filtrati sotto le 3 lettere: una forma corta
    troppo generica ("V", "Co") darebbe falsi positivi su qualsiasi titolo.
    """
    grezzi = [company_name, *aliases]
    corte = [_spoglia_suffisso_societario(n) for n in grezzi]
    viste: dict[str, None] = {}
    for nome in [*grezzi, *corte]:
        nome = nome.strip()
        if len(nome) >= 3:
            viste.setdefault(nome, None)
    return tuple(viste.keys())
```

```python
@dataclass(frozen=True)
class VerdictoSecondoOrdine:
    secondo_ordine: bool
    connettore: str | None = None
    terza_azienda_ticker: str | None = None
    terza_azienda_nome: str | None = None


# Precisione prima di recall: solo connettori causali forti. "as"/"on" da
# soli sono troppo comuni e produrrebbero falsi positivi diffusi.
_CONNETTORI = (
    "following",
    "after",
    "on the back of",
    "amid",
    "on news of",
    "thanks to",
    "as a result of",
)


def _cerca_nome(testo_lower: str, azienda: Azienda) -> str | None:
    for nome in azienda.nomi:
        if re.search(r"\b" + re.escape(nome.lower()) + r"\b", testo_lower):
            return nome
    return None
```

Add after `costruisci_aziende`:

```python
def classifica_secondo_ordine(
    titolo: str,
    proprio_ticker: str,
    aziende: Mapping[str, Azienda],
) -> VerdictoSecondoOrdine:
    """Il titolo nomina la NOSTRA azienda, con un connettore causale che
    introduce una azienda DIVERSA nota in `ticker_lookup`.

    Tre condizioni, tutte necessarie:
    1. auto-riferimento — il titolo nomina l'azienda di `proprio_ticker`
       (altrimenti e' fuori scope: e' un difetto di tagging, #405, non di
       magnitudine sotto-scorata, #408);
    2. un connettore causale forte compare nel titolo;
    3. dopo il connettore compare un'azienda NOTA diversa dalla nostra.

    Il connettore piu' a sinistra nel titolo vince, indipendentemente
    dall'ordine della lista `_CONNETTORI`: e' quello grammaticalmente piu'
    vicino al soggetto ("Adobe ... following ... Salesforce").
    """
    propria = aziende.get(proprio_ticker)
    if propria is None:
        return VerdictoSecondoOrdine(False)

    titolo_lower = titolo.lower()
    if _cerca_nome(titolo_lower, propria) is None:
        return VerdictoSecondoOrdine(False)

    migliore: tuple[str, re.Match[str]] | None = None
    for connettore in _CONNETTORI:
        m = re.search(r"\b" + re.escape(connettore) + r"\b", titolo_lower)
        if m and (migliore is None or m.start() < migliore[1].start()):
            migliore = (connettore, m)
    if migliore is None:
        return VerdictoSecondoOrdine(False)
    connettore_trovato, m = migliore

    coda = titolo_lower[m.end():]
    for ticker, azienda in aziende.items():
        if ticker == proprio_ticker:
            continue
        nome_trovato = _cerca_nome(coda, azienda)
        if nome_trovato is not None:
            return VerdictoSecondoOrdine(
                secondo_ordine=True,
                connettore=connettore_trovato,
                terza_azienda_ticker=ticker,
                terza_azienda_nome=nome_trovato,
            )

    return VerdictoSecondoOrdine(False)
```

```python
def _statistiche(valori: list[float]) -> dict[str, float | int | None]:
    if not valori:
        return {"media": None, "mediana": None, "n": 0}
    ordinati = sorted(valori)
    n = len(ordinati)
    mediana = (
        ordinati[n // 2]
        if n % 2 == 1
        else (ordinati[n // 2 - 1] + ordinati[n // 2]) / 2
    )
    return {"media": sum(valori) / n, "mediana": mediana, "n": n}


def aggrega_risultati(righe: Iterable[RigaClassificata]) -> dict:
    """Conteggio, tasso, distribuzione dei ritorni attesi per
    secondo_ordine vs il resto, e tasso di accordo col self-report del
    modello (`directness`). Nessuna affermazione causale: solo confronto."""
    righe = list(righe)
    totale = len(righe)
    secondo_ordine_righe = [r for r in righe if r.verdetto.secondo_ordine]
    n_secondo_ordine = len(secondo_ordine_righe)

    ritorni_attesi = {}
    for etichetta, campo in (
        ("1g", "forward_return"),
        ("3g", "forward_return_3d"),
        ("5g", "forward_return_5d"),
    ):
        valori_so = [
            getattr(r, campo) for r in righe
            if r.verdetto.secondo_ordine and getattr(r, campo) is not None
        ]
        valori_altri = [
            getattr(r, campo) for r in righe
            if not r.verdetto.secondo_ordine and getattr(r, campo) is not None
        ]
        ritorni_attesi[etichetta] = {
            "secondo_ordine": _statistiche(valori_so),
            "altri": _statistiche(valori_altri),
        }

    con_directness = [r for r in secondo_ordine_righe if r.directness_modelli]
    concordi = [
        r for r in con_directness
        if any(d in _CANALI_SPILLOVER_MODELLO for d in r.directness_modelli)
    ]

    return {
        "totale_articoli": totale,
        "secondo_ordine": {
            "conteggio": n_secondo_ordine,
            "tasso": (n_secondo_ordine / totale) if totale else None,
        },
        "ritorni_attesi": ritorni_attesi,
        "accordo_modello": {
            "n_con_directness": len(con_directness),
            "concorde": len(concordi),
            "tasso_accordo": (
                len(concordi) / len(con_directness) if con_directness else None
            ),
        },
    }
```

```python
#!/usr/bin/env python3
"""Quante notizie di secondo ordine/spillover ci sono, e come si muove il
prezzo dopo (#408, punto 1 dell'investigation — il resto e' gia' stato
deployato come Variante A del prompt, #399/#408, `bf5bef2e`).

indipendente dal campo `directness` auto-riportato dal modello — vedi il
modulo per il perche'.

Uso:
    export DATABASE_URL=postgresql://trading:trading@localhost:5432/trading
    uv run python scripts/characterize_second_order_news.py
"""
from __future__ import annotations

import json
from datetime import datetime, timezone
from pathlib import Path

from src.analysis.second_order_news import (
    RigaClassificata,
    aggrega_risultati,
    classifica_secondo_ordine,
    costruisci_aziende,
)

_PERCORSO_EVIDENZA = Path(__file__).resolve().parents[1] / "docs" / "evidence" / "second_order_news.json"

_QUERY_TICKER_LOOKUP = "SELECT ticker, company_name, aliases FROM ticker_lookup"

_QUERY_NEWS = """
    SELECT
        nl.id, nl.ticker, nl.title,
        ss.forward_return, ss.forward_return_3d, ss.forward_return_5d,
        array_remove(array_agg(DISTINCT lr.directness), NULL) AS directness_modelli
    FROM news_log nl
    LEFT JOIN sentiment_signals ss ON ss.news_log_id = nl.id
    LEFT JOIN llm_responses lr ON lr.signal_id = ss.id
    GROUP BY nl.id, nl.ticker, nl.title, ss.id, ss.forward_return,
             ss.forward_return_3d, ss.forward_return_5d
"""


def _connetti():
    import psycopg2

    from src.config import config

    return psycopg2.connect(config.DATABASE_URL)


def main() -> int:
    with _connetti() as conn:
        with conn.cursor() as cur:
            cur.execute(_QUERY_TICKER_LOOKUP)
            righe_ticker_lookup = cur.fetchall()
        with conn.cursor() as cur:
            cur.execute(_QUERY_NEWS)
            righe_news = cur.fetchall()

    aziende = costruisci_aziende(righe_ticker_lookup)

    righe_classificate: list[RigaClassificata] = []
    esempi: list[dict] = []
    for news_log_id, ticker, titolo, fr1, fr3, fr5, directness_modelli in righe_news:
        verdetto = classifica_secondo_ordine(titolo, ticker, aziende)
        righe_classificate.append(
            RigaClassificata(verdetto, fr1, fr3, fr5, tuple(directness_modelli or ()))
        )
        if verdetto.secondo_ordine and len(esempi) < 20:
            esempi.append({
                "news_log_id": news_log_id,
                "ticker": ticker,
                "titolo": titolo,
                "connettore": verdetto.connettore,
                "terza_azienda_ticker": verdetto.terza_azienda_ticker,
                "terza_azienda_nome": verdetto.terza_azienda_nome,
                "forward_return": fr1,
            })

    sintesi = aggrega_risultati(righe_classificate)

    evidenza = {
        "generato_il": datetime.now(timezone.utc).isoformat(),
        "issue": "#408",
        "metodo": (
            "rilevatore deterministico (src.analysis.second_order_news): "
            "auto-riferimento alla nostra azienda + connettore causale + "
            "azienda terza nota in ticker_lookup dopo il connettore. Una "
            "riga per (news_log, sentiment_signals) — non usa il campo "
            "directness auto-riportato dal modello come fonte della verita'."
        ),
        "sintesi": sintesi,
        "esempi": esempi,
    }

    _PERCORSO_EVIDENZA.parent.mkdir(parents=True, exist_ok=True)
    _PERCORSO_EVIDENZA.write_text(json.dumps(evidenza, indent=2, ensure_ascii=False))
    print(f"Scritto {_PERCORSO_EVIDENZA} — {sintesi['totale_articoli']} articoli, "
          f"{sintesi['secondo_ordine']['conteggio']} secondo_ordine "
          f"({sintesi['secondo_ordine']['tasso']}).")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
